cmake_minimum_required(VERSION 3.16)
project(cpp_jsonlite LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(test_jsonlite tests/test_jsonlite.cpp)
target_include_directories(test_jsonlite PRIVATE include)
if(MSVC)
  target_compile_options(test_jsonlite PRIVATE /W4)
else()
  target_compile_options(test_jsonlite PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME test_jsonlite COMMAND test_jsonlite)
